%% institucion.tex — Datos de la universidad y formato de las portadas.
%% Un comando por línea. Todos son opcionales: si uno queda vacío, { }, no
%% sale en las portadas. Rellena los de tu universidad; si tu centro pide un
%% formato concreto (márgenes, tamaño de letra…), ajústalo aquí.

\universidad{}                            % p. ej. Universidad de …
\escuela{}                                % escuela o facultad
\logo{}                                   % archivo dentro de estilo/ (.pdf, .png o .jpg); vacío = sin logo
\tipoTrabajo{tfg}                         % tfg | tfm | tesis | otro
\nombreTrabajo{}                          % (opcional) sustituye a «Trabajo Fin de Grado»…
\titulacion{}                             % p. ej. Grado en Ingeniería Informática
\etiquetaEspecialidad{}                   % p. ej. Mención; vacío = no se muestra la especialidad
\idiomaPortadas{documento}                % espanol | ingles | documento (el de \idioma en datos.tex)

%% Formato de página
\tamanoLetra{12pt}                        % 10pt | 11pt | 12pt
\margenes{30mm}{25mm}{25mm}{25mm}         % interior (encuadernación), exterior, superior, inferior
\interlineado{1.5}                        % 1 | 1.15 | 1.25 | 1.5 | 2
